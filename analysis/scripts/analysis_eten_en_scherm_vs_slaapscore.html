analyse_data <- data

groeps_kleuren <- c("persoon 1" = "#F8766D",
                    "persoon 2" = "#00BA38",
                    "persoon 3" = "#619CFF")

5. Analyse A — Eten-timing vs Gemiddelde Hartslag tijdens Slaap

Correlatie

analyse_data %>%
  group_by(naam) %>%
  summarise(
    R2 = round(summary(lm(average_sleep_heart_rate ~ time_between_last_food_and_sleep_hours))$r.squared, 3),
    p  = round(cor.test(time_between_last_food_and_sleep_hours, average_sleep_heart_rate)$p.value, 3)
  ) %>%
  print()
## # A tibble: 3 × 3

Plot — alle personen vergeleken

ggplot(data, aes(x = time_between_last_food_and_sleep_hours, y = average_sleep_heart_rate,
                          color = naam, fill = naam)) +
  geom_point(size = 3.5, alpha = 0.8, na.rm = TRUE) +
  geom_smooth(method = "lm", alpha = 0.15, na.rm = TRUE) +
  scale_color_manual(values = groeps_kleuren, name = "Persoon") +
  scale_fill_manual(values  = groeps_kleuren, name = "Persoon") +
  labs(
    title = "Alle personen — Uren tussen laatste maaltijd en slaap vs Gemiddelde Hartslag",
    x     = "Aantal uren tussen eten en slapen",
    y     = "Gemiddelde Hartslag tijdens Slaap (bpm)"
  ) +
  theme_minimal()
