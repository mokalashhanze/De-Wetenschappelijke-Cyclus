1. Libraries laden

library(ggplot2)

2. Data inladen

sleep_data     <- read.csv("/Users/robinoffringa/Desktop/De-Wetenschappelijke-Cyclus/takeout_robin/Fitbit/Sleep Score/sleep_score.csv")
heartrate_data <- read.csv("/Users/robinoffringa/Desktop/De-Wetenschappelijke-Cyclus/takeout_robin/Fitbit/average_heart_rate_excluding_sleep.csv")

3. Data voorbereiden & samenvoegen

# Datumkolommen omzetten
sleep_data$date     <- as.Date(sleep_data$timestamp)
heartrate_data$date <- as.Date(heartrate_data$date)

# Merge op datum
data <- merge(
  sleep_data[, c("date", "overall_score", "resting_heart_rate")],
  heartrate_data[, c("date", "average_heart_rate", "total_steps")],
  by = "date"
)

# Verwijder rijen met ontbrekende waarden
data <- data[complete.cases(data), ]

cat("Data punten: n =", nrow(data), "\n")
## Data punten: n = 6

4. Statistieken

cat("Slaapscore       — Mean:", round(mean(data$overall_score), 2),
    ", SD:", round(sd(data$overall_score), 2), "\n")
## Slaapscore       — Mean: 75.67 , SD: 9.37
cat("Hartslag overdag — Mean:", round(mean(data$average_heart_rate), 2),
    ", SD:", round(sd(data$average_heart_rate), 2), "\n")
## Hartslag overdag — Mean: 75.18 , SD: 4.23
cat("Stappen per dag   — Mean:", round(mean(data$total_steps), 2),
  ", SD:", round(sd(data$total_steps), 2), "\n")
## Stappen per dag   — Mean: 11284.83 , SD: 2039.82

5. Analyse A — Slaapscore vs Hartslag Overdag

Correlatie

corr <- cor.test(data$overall_score, data$average_heart_rate, alternative = "two.sided")
cat("r =", round(corr$estimate, 3), "\n")
## r = -0.493
cat("p =", round(corr$p.value, 3), "\n")
## p = 0.321

Plot

ggplot(data, aes(x = average_heart_rate, y = overall_score)) +
  geom_point(size = 4, color = "blue", alpha = 0.7) +
  geom_smooth(method = "lm", color = "red") +
  labs(
    title = "Slaapscore vs Hartslag Overdag",
    x     = "Gemiddelde Hartslag (bpm)",
    y     = "Slaapscore"
  ) +
  theme_minimal()
## `geom_smooth()` using formula = 'y ~ x'
